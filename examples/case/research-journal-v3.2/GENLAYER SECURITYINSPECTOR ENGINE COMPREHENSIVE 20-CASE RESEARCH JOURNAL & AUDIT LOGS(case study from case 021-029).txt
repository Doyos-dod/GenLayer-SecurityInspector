```python
# { "Depends": "py-genlayer:case-025" }

import os
import random
import genlayer as gl

class FullSpectrumCommon(gl.Contract):

    owner: str
    balance: int
    status: str
    trail: str
    records: list
    cache: dict

    def __init__(self):
        self.owner = "INITIAL_OWNER"
        self.balance = 1000
        self.status = "IDLE"
        self.trail = ""
        self.records = []
        self.cache = {}

    def _external_peer(self):
        peer = gl.contract.get_at(
            "0x1234567890"
        )
        return gl.call(peer)

    def _nondeterministic_source(self):
        return gl.vm.run_nondet_default(
            lambda: random.randint(1, 100)
        )

    def _get_ai_decision(self):
        first = gl.nondet.llm(
            lambda: "APPROVE"
        )
        second = gl.nondet.llm(
            lambda: "TRANSFER"
        )
        third = gl.nondet.llm(
            lambda: "EXECUTE"
        )
        return first + second + third

    @gl.public.write
    def execute(self, amount: int):

        # Access control is intentionally absent.
        self.balance -= amount

        # Time dependencies.
        if now() > 100000:
            self.status = "ACTIVE"

        if created() > 100000:
            self.status = "TIME_LOCKED"

        # Unbounded storage growth.
        self.trail += str(amount)

        # External contract response.
        peer_result = self._external_peer()

        # External web response.
        web_result = gl.nondet.web.get(
            "https://example.com/data"
        )

        rendered_result = gl.nondet.web.render(
            "https://example.com/page"
        )

        # AI output is used without semantic validation.
        ai_result = self._get_ai_decision()

        if ai_result:
            self.status = str(ai_result)

        self.cache["peer"] = peer_result
        self.cache["web"] = web_result
        self.records.append(rendered_result)

        self.balance = self.balance + self._nondeterministic_source()

    @gl.public.write
    def emergency_reset(self):
        # Another deliberately unprotected write method.
        self.balance = 0
        self.status = "RESET"

    @gl.public.view
    def get_status(self):
        return self.status
```

CASE-026 — ADVERSARIAL: OBFUSCATED API RESOLUTION
Dynamic Attribute Access & SEC011 Blind Spot

OBJECTIVE
Test whether SecurityInspector v3.2 detects security-relevant API usage
when GenLayer API attributes are resolved dynamically through getattr()
instead of appearing as direct literal call expressions.

TEST SOURCE CODE (AS SUBMITTED)

```python
# { "Depends": "py-genlayer:case-026" }

import genlayer as gl


class DynamicResolverProbe(gl.Contract):

    owner: str
    status: str
    result: str

    def __init__(self):
        self.owner = "INITIAL_OWNER"
        self.status = "IDLE"
        self.result = ""

    def _resolve_peer(self):
        contract_module = getattr(gl, "contract")
        lookup_function = getattr(contract_module, "get_at")
        peer = lookup_function("0x1234567890")

        invoke_function = getattr(gl, "call")
        return invoke_function(peer)

    def _resolve_ai(self):
        nondet_module = getattr(gl, "nondet")
        model_function = getattr(nondet_module, "llm")

        return model_function(
            lambda: "APPROVE"
        )

    def _resolve_web(self):
        nondet_module = getattr(gl, "nondet")
        web_module = getattr(nondet_module, "web")
        fetch_function = getattr(web_module, "get")

        return fetch_function(
            "https://example.com/data"
        )
```
